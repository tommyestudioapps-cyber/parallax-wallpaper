── PARTE 1: Criar o plugin ──
Crie o arquivo artifacts/parallax-wallpaper/plugins/withKotlinVersion.js:

  const fs = require('fs');
  const path = require('path');
  const { withDangerousMod } = require('expo/config-plugins');

  // Workaround: o expo-build-properties escreve android.kotlinVersion em
  // gradle.properties, mas o Kotlin Gradle Plugin neste setup (Expo SDK 54
  // + RN 0.81) ignora a property. Injetamos a versão explícita no classpath.
  // Remover quando o ecossistema passar a consumir a property corretamente.
  const KOTLIN_VERSION = '2.2.0';

  module.exports = function withKotlinVersion(config) {
    return withDangerousMod(config, [
      'android',
      async (config) => {
        const buildGradlePath = path.join(
          config.modRequest.platformProjectRoot,
          'build.gradle',
        );
        if (!fs.existsSync(buildGradlePath)) {
          throw new Error('withKotlinVersion: android/build.gradle não encontrado');
        }
        let content = fs.readFileSync(buildGradlePath, 'utf8');

        // Substitui a linha sem versão pela linha com versão explícita.
        const before = /classpath\((['"])org\.jetbrains\.kotlin:kotlin-gradle-plugin\1\)/;
        const after = `classpath('org.jetbrains.kotlin:kotlin-gradle-plugin:${KOTLIN_VERSION}')`;
        if (!before.test(content)) {
          throw new Error(
            'withKotlinVersion: classpath do kotlin-gradle-plugin não encontrado em build.gradle',
          );
        }
        content = content.replace(before, after);
        fs.writeFileSync(buildGradlePath, content);

        // Sanidade: confirmar que sobrou apenas uma ocorrência com versão.
        if (!content.includes(`kotlin-gradle-plugin:${KOTLIN_VERSION}`)) {
          throw new Error('withKotlinVersion: injeção falhou');
        }
        return config;
      },
    ]);
  };
