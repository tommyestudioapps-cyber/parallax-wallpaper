A Engine EGL Nativa (WallpaperService.Engine)Para que um Live Wallpaper funcione corretamente sem depender de GLSurfaceView (que não se integra nativamente ao ciclo de vida da superfície do sistema), construímos uma thread OpenGL dedicada com EGL 1.4:Kotlinimport android.opengl.EGL14
import android.opengl.EGLConfig
import android.opengl.EGLContext
import android.opengl.EGLDisplay
import android.opengl.EGLSurface
import android.opengl.GLES20
import android.service.wallpaper.WallpaperService
import android.view.SurfaceHolder

abstract class OpenGLEngineService : WallpaperService() {

    abstract inner class GLEngine : Engine(), Runnable {

        @Volatile
        private var isRunning = false
        private var renderThread: Thread? = null

        private var eglDisplay: EGLDisplay = EGL14.EGL_NO_DISPLAY
        private var eglContext: EGLContext = EGL14.EGL_NO_CONTEXT
        private var eglSurface: EGLSurface = EGL14.EGL_NO_SURFACE

        private var surfaceWidth = 0
        private var surfaceHeight = 0

        override fun onVisibilityChanged(visible: Boolean) {
            super.onVisibilityChanged(visible)
            if (visible) {
                startRenderThread()
            } else {
                stopRenderThread()
            }
        }

        override fun onSurfaceCreated(holder: SurfaceHolder) {
            super.onSurfaceCreated(holder)
            startRenderThread()
        }

        override fun onSurfaceChanged(holder: SurfaceHolder, format: Int, width: Int, height: Int) {
            super.onSurfaceChanged(holder, format, width, height)
            surfaceWidth = width
            surfaceHeight = height
        }

        override fun onSurfaceDestroyed(holder: SurfaceHolder) {
            super.onSurfaceDestroyed(holder)
            stopRenderThread()
        }

        @Synchronized
        private fun startRenderThread() {
            if (!isRunning && surfaceHolder.surface.isValid) {
                isRunning = true
                renderThread = Thread(this, "GLParallaxRenderThread").apply { start() }
            }
        }

        @Synchronized
        private fun stopRenderThread() {
            isRunning = false
            renderThread?.join(500)
            renderThread = null
        }

        override fun run() {
            initEGL()
            onGLCreated()

            var lastFrameTime = System.nanoTime()

            while (isRunning) {
                // Controla o frame rate (~60 FPS / VSync) para evitar consumo excessivo da GPU
                val now = System.nanoTime()
                val targetFrameTime = 1_000_000_000L / 60L // 16.6ms
                val elapsed = now - lastFrameTime

                if (elapsed < targetFrameTime) {
                    try {
                        Thread.sleep((targetFrameTime - elapsed) / 1_000_000L)
                    } catch (e: InterruptedException) {
                        break
                    }
                }
                lastFrameTime = System.nanoTime()

                if (surfaceWidth > 0 && surfaceHeight > 0) {
                    GLES20.glViewport(0, 0, surfaceWidth, surfaceHeight)
                    onGLDraw(surfaceWidth, surfaceHeight)
                    EGL14.eglSwapBuffers(eglDisplay, eglSurface)
                }
            }

            onGLDestroy()
            destroyEGL()
        }

        private fun initEGL() {
            eglDisplay = EGL14.eglGetDisplay(EGL14.EGL_DEFAULT_DISPLAY)
            val version = IntArray(2)
            EGL14.eglInitialize(eglDisplay, version, 0, version, 1)

            val configAttribs = intArrayOf(
                EGL14.EGL_RENDERABLE_TYPE, EGL14.EGL_OPENGL_ES2_BIT,
                EGL14.EGL_RED_SIZE, 8,
                EGL14.EGL_GREEN_SIZE, 8,
                EGL14.EGL_BLUE_SIZE, 8,
                EGL14.EGL_ALPHA_SIZE, 8,
                EGL14.EGL_NONE
            )

            val configs = arrayOfNulls<EGLConfig>(1)
            val numConfigs = IntArray(1)
            EGL14.eglChooseConfig(eglDisplay, configAttribs, 0, configs, 0, 1, numConfigs, 0)

            val contextAttribs = intArrayOf(
                EGL14.EGL_CONTEXT_CLIENT_VERSION, 2,
                EGL14.EGL_NONE
            )

            eglContext = EGL14.eglCreateContext(eglDisplay, configs[0], EGL14.EGL_NO_CONTEXT, contextAttribs, 0)
            eglSurface = EGL14.eglCreateWindowSurface(eglDisplay, configs[0], surfaceHolder.surface, intArrayOf(EGL14.EGL_NONE), 0)
            EGL14.eglMakeCurrent(eglDisplay, eglSurface, eglSurface, eglContext)
        }

        private fun destroyEGL() {
            if (eglDisplay != EGL14.EGL_NO_DISPLAY) {
                EGL14.eglMakeCurrent(eglDisplay, EGL14.EGL_NO_SURFACE, EGL14.EGL_NO_SURFACE, EGL14.EGL_NO_CONTEXT)
                if (eglSurface != EGL14.EGL_NO_SURFACE) EGL14.eglDestroySurface(eglDisplay, eglSurface)
                if (eglContext != EGL14.EGL_NO_CONTEXT) EGL14.eglDestroyContext(eglDisplay, eglContext)
                EGL14.eglTerminate(eglDisplay)
            }
            eglDisplay = EGL14.EGL_NO_DISPLAY
            eglContext = EGL14.EGL_NO_CONTEXT
            eglSurface = EGL14.EGL_NO_SURFACE
        }

        abstract fun onGLCreated()
        abstract fun onGLDraw(width: Int, height: Int)
        abstract fun onGLDestroy()
    }
}
2. Gerenciador de Ciclo de Vida de Texturas (TextureManager.kt)Resolve o problema do vazamento de VRAM, garantindo a liberação explícita via glDeleteTextures quando imagens são trocadas ou o contexto é destruído:Kotlinimport android.graphics.Bitmap
import android.opengl.GLES20
import android.opengl.GLUtils

class TextureManager {

    private val textureIds = IntArray(1)

    /**
     * Sobe um bitmap para a GPU e deleta a textura antiga associada se ela existir.
     */
    fun uploadOrReplaceTexture(oldTextureId: Int, bitmap: Bitmap?): Int {
        if (bitmap == null || bitmap.isRecycled) {
            deleteTexture(oldTextureId)
            return 0
        }

        // Deleta a textura antiga antes de criar uma nova no contexto GL
        deleteTexture(oldTextureId)

        // Assegura que o bitmap foi decodificado como Premultiplied Alpha
        if (!bitmap.isPremultiplied) {
            bitmap.isPremultiplied = true
        }

        GLES20.glGenTextures(1, textureIds, 0)
        val newTextureId = textureIds[0]

        GLES20.glBindTexture(GLES20.GL_TEXTURE_2D, newTextureId)
        GLES20.glTexParameteri(GLES20.GL_TEXTURE_2D, GLES20.GL_TEXTURE_MIN_FILTER, GLES20.GL_LINEAR)
        GLES20.glTexParameteri(GLES20.GL_TEXTURE_2D, GLES20.GL_TEXTURE_MAG_FILTER, GLES20.GL_LINEAR)
        GLES20.glTexParameteri(GLES20.GL_TEXTURE_2D, GLES20.GL_TEXTURE_WRAP_S, GLES20.GL_CLAMP_TO_EDGE)
        GLES20.glTexParameteri(GLES20.GL_TEXTURE_2D, GLES20.GL_TEXTURE_WRAP_T, GLES20.GL_CLAMP_TO_EDGE)

        GLUtils.texImage2D(GLES20.GL_TEXTURE_2D, 0, bitmap, 0)
        
        return newTextureId
    }

    fun deleteTexture(textureId: Int) {
        if (textureId > 0) {
            textureIds[0] = textureId
            GLES20.glDeleteTextures(1, textureIds, 0)
        }
    }
}

4. Implementação Completa do Parallax ShaderEsta classe consolida o compilador de shaders, a inicialização do quad fullscreen, a configuração dos bindings e samplers de textura e a renderização a zero-allocation:Kotlinimport android.graphics.Bitmap
import android.opengl.GLES20
import java.nio.ByteBuffer
import java.nio.ByteOrder
import java.nio.FloatBuffer

class ParallaxGLRenderer {

    private var programHandle = -1

    // Textura Handles
    var texBackId = 0
    var texMiddleId = 0
    var texForeId = 0

    // Uniform handles
    private var uTexBackHandle = -1
    private var uTexMiddleHandle = -1
    private var uTexForeHandle = -1
    private var uMatBackHandle = -1
    private var uMatMiddleHandle = -1
    private var uMatForeHandle = -1
    private var uAlphaMiddleHandle = -1
    private var uAlphaForeHandle = -1

    // Helpers de Matriz
    val backMatrix = ParallaxMatrixHelper()
    val middleMatrix = ParallaxMatrixHelper()
    val foreMatrix = ParallaxMatrixHelper()

    private val textureManager = TextureManager()
    private var vertexBuffer: FloatBuffer? = null

    // Vértices do Quad Fullscreen e UVs (Triangle Strip)
    private val quadData = floatArrayOf(
        // Pos(X, Y)  UV(S, T)
        -1.0f,  1.0f, 0.0f, 0.0f,
        -1.0f, -1.0f, 0.0f, 1.0f,
         1.0f,  1.0f, 1.0f, 0.0f,
         1.0f, -1.0f, 1.0f, 1.0f
    )

    fun onCreated() {
        // 1. Cria o Vertex Buffer
        vertexBuffer = ByteBuffer.allocateDirect(quadData.size * 4)
            .order(ByteOrder.nativeOrder())
            .asFloatBuffer().apply {
                put(quadData)
                position(0)
            }

        // 2. Compila Shaders
        val vertexShader = loadShader(GLES20.GL_VERTEX_SHADER, VERTEX_SHADER_CODE)
        val fragmentShader = loadShader(GLES20.GL_FRAGMENT_SHADER, FRAGMENT_SHADER_CODE)

        programHandle = GLES20.glCreateProgram().also {
            GLES20.glAttachShader(it, vertexShader)
            GLES20.glAttachShader(it, fragmentShader)
            GLES20.glLinkProgram(it)
        }

        // 3. Localiza os Uniforms e Atributos
        uTexBackHandle = GLES20.glGetUniformLocation(programHandle, "u_TexBackground")
        uTexMiddleHandle = GLES20.glGetUniformLocation(programHandle, "u_TexMiddle")
        uTexForeHandle = GLES20.glGetUniformLocation(programHandle, "u_TexForeground")

        uMatBackHandle = GLES20.glGetUniformLocation(programHandle, "u_MatBack")
        uMatMiddleHandle = GLES20.glGetUniformLocation(programHandle, "u_MatMiddle")
        uMatForeHandle = GLES20.glGetUniformLocation(programHandle, "u_MatFore")

        uAlphaMiddleHandle = GLES20.glGetUniformLocation(programHandle, "u_AlphaMiddle")
        uAlphaForeHandle = GLES20.glGetUniformLocation(programHandle, "u_AlphaFore")
    }

    fun updateBitmaps(back: Bitmap?, middle: Bitmap?, fore: Bitmap?) {
        texBackId = textureManager.uploadOrReplaceTexture(texBackId, back)
        texMiddleId = textureManager.uploadOrReplaceTexture(texMiddleId, middle)
        texForeId = textureManager.uploadOrReplaceTexture(texForeId, fore)
    }

    fun onDraw() {
        GLES20.glUseProgram(programHandle)

        // Configuração dos atributos do Quad
        vertexBuffer?.let { buffer ->
            val stride = 4 * 4 // 4 floats por vértice
            
            buffer.position(0)
            val aPosHandle = GLES20.glGetAttribLocation(programHandle, "a_Position")
            GLES20.glEnableVertexAttribArray(aPosHandle)
            GLES20.glVertexAttribPointer(aPosHandle, 2, GLES20.GL_FLOAT, false, stride, buffer)

            buffer.position(2)
            val aTexCoordHandle = GLES20.glGetAttribLocation(programHandle, "a_TexCoord")
            GLES20.glEnableVertexAttribArray(aTexCoordHandle)
            GLES20.glVertexAttribPointer(aTexCoordHandle, 2, GLES20.GL_FLOAT, false, stride, buffer)
        }

        // Binding das 3 Unidades de Textura
        GLES20.glActiveTexture(GLES20.GL_TEXTURE0)
        GLES20.glBindTexture(GLES20.GL_TEXTURE_2D, texBackId)
        GLES20.glUniform1i(uTexBackHandle, 0)

        GLES20.glActiveTexture(GLES20.GL_TEXTURE1)
        GLES20.glBindTexture(GLES20.GL_TEXTURE_2D, texMiddleId)
        GLES20.glUniform1i(uTexMiddleHandle, 1)

        GLES20.glActiveTexture(GLES20.GL_TEXTURE2)
        GLES20.glBindTexture(GLES20.GL_TEXTURE_2D, texForeId)
        GLES20.glUniform1i(uTexForeHandle, 2)

        // Envia as Matrizes dos Buffers Persistentes (Zero-Allocation)
        GLES20.glUniformMatrix3fv(uMatBackHandle, 1, false, backMatrix.matrixBuffer)
        GLES20.glUniformMatrix3fv(uMatMiddleHandle, 1, false, middleMatrix.matrixBuffer)
        GLES20.glUniformMatrix3fv(uMatForeHandle, 1, false, foreMatrix.matrixBuffer)

        GLES20.glUniform1f(uAlphaMiddleHandle, 1.0f)
        GLES20.glUniform1f(uAlphaForeHandle, 1.0f)

        // Desenho de uma única passagem
        GLES20.glDrawArrays(GLES20.GL_TRIANGLE_STRIP, 0, 4)
    }

    fun onDestroy() {
        textureManager.deleteTexture(texBackId)
        textureManager.deleteTexture(texMiddleId)
        textureManager.deleteTexture(texForeId)
        if (programHandle != -1) {
            GLES20.glDeleteProgram(programHandle)
        }
    }

    private fun loadShader(type: Int, shaderCode: String): Int {
        return GLES20.glCreateShader(type).also { shader ->
            GLES20.glShaderSource(shader, shaderCode)
            GLES20.glCompileShader(shader)
        }
    }

    companion object {
        private const val VERTEX_SHADER_CODE = """
            attribute vec4 a_Position;
            attribute vec2 a_TexCoord;
            varying vec2 v_TexCoord;
            void main() {
                gl_Position = a_Position;
                v_TexCoord = a_TexCoord;
            }
        """

        private const val FRAGMENT_SHADER_CODE = """
            #ifdef GL_FRAGMENT_PRECISION_HIGH
            precision highp float;
            #else
            precision mediump float;
            #endif

            uniform sampler2D u_TexBackground;
            uniform sampler2D u_TexMiddle;
            uniform sampler2D u_TexForeground;

            uniform mat3 u_MatBack;
            uniform mat3 u_MatMiddle;
            uniform mat3 u_MatFore;

            uniform float u_AlphaMiddle;
            uniform float u_AlphaFore;

            varying vec2 v_TexCoord;

            vec4 sampleCutoutLayer(sampler2D tex, mat3 transformMat, float layerAlpha) {
                if (layerAlpha <= 0.001) return vec4(0.0);
                vec3 transformedUV = transformMat * vec3(v_TexCoord, 1.0);
                vec2 uv = transformedUV.xy;
                if (uv.x < 0.0 || uv.x > 1.0 || uv.y < 0.0 || uv.y > 1.0) return vec4(0.0);
                return texture2D(tex, uv) * layerAlpha;
            }

            vec4 blendPremultiplied(vec4 back, vec4 front) {
                return front + back * (1.0 - front.a);
            }

            void main() {
                vec3 uvBack = u_MatBack * vec3(v_TexCoord, 1.0);
                vec4 cBack = texture2D(u_TexBackground, clamp(uvBack.xy, 0.0, 1.0));
                cBack.a = 1.0;

                vec4 cMiddle = sampleCutoutLayer(u_TexMiddle, u_MatMiddle, u_AlphaMiddle);
                vec4 cFore   = sampleCutoutLayer(u_TexForeground, u_MatFore, u_AlphaFore);

                vec4 composite = blendPremultiplied(cBack, cMiddle);
                composite      = blendPremultiplied(composite, cFore);

                gl_FragColor = vec4(composite.rgb, 1.0);
            }
        """
    }
}
