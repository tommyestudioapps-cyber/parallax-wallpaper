import java.nio.ByteBuffer
import java.nio.ByteOrder
import java.nio.FloatBuffer

class ParallaxMatrixHelper {

    // Buffer fixo nativo alocado UMA ÚNICA VEZ na memória para passar à GPU (3x3 floats = 36 bytes)
    val matrixBuffer: FloatBuffer = ByteBuffer.allocateDirect(9 * 4)
        .order(ByteOrder.nativeOrder())
        .asFloatBuffer()

    // Array de apoio persistente para evitar criação de arrays a 60 FPS
    private val glMatrixValues = FloatArray(9)

    /**
     * Calcula a matriz Tela -> UV da Textura diretamente e atualiza o buffer persistente.
     * 
     * @param cropX Pixel X da imagem que deve ficar no centro da tela
     * @param cropY Pixel Y da imagem que deve ficar no centro da tela
     * @param parallaxOffsetX Deslocamento do giroscópio em pixels da imagem
     * @param parallaxOffsetY Deslocamento do giroscópio em pixels da imagem
     */
    fun updateLayerMatrix(
        viewWidth: Float, viewHeight: Float,
        imgWidth: Float, imgHeight: Float,
        scale: Float, cropX: Float, cropY: Float,
        parallaxOffsetX: Float, parallaxOffsetY: Float,
        isBackground: Boolean = false
    ) {
        // 1. Validação de segurança contra divisão por zero/matrizes degeneradas
        if (viewWidth <= 0f || viewHeight <= 0f || imgWidth <= 0f || imgHeight <= 0f || scale <= 0f) {
            setIdentity()
            return
        }

        // Aplica o Overscan de 10% no fundo para evitar bordas durante o movimento
        val overscan = if (isBackground) 1.10f else 1.0f
        val finalScale = scale * overscan

        // Coeficientes da escala
        val sx = viewWidth / (imgWidth * finalScale)
        val sy = viewHeight / (imgHeight * finalScale)

        // Soma os deslocamentos de Crop e Giroscópio (em pixels da imagem)
        val totalOffsetX = cropX + parallaxOffsetX
        val totalOffsetY = cropY + parallaxOffsetY

        // Coeficientes de translação alinhados ao centro da textura (0.5, 0.5)
        val tx = 0.5f - (viewWidth / (2f * imgWidth * finalScale)) - (totalOffsetX / imgWidth)
        val ty = 0.5f - (viewHeight / (2f * imgHeight * finalScale)) - (totalOffsetY / imgHeight)

        // Montagem da Matriz 3x3 em ordem Column-Major (OpenGL ES)
        // [ sx,   0,   0 ]
        // [  0,  sy,   0 ]
        // [ tx,  ty,   1 ]
        glMatrixValues[0] = sx
        glMatrixValues[1] = 0f
        glMatrixValues[2] = 0f

        glMatrixValues[3] = 0f
        glMatrixValues[4] = sy
        glMatrixValues[5] = 0f

        glMatrixValues[6] = tx
        glMatrixValues[7] = ty
        glMatrixValues[8] = 1f

        // Copia os valores para o Buffer nativo sem realocar memória
        matrixBuffer.clear()
        matrixBuffer.put(glMatrixValues)
        matrixBuffer.position(0)
    }

    private fun setIdentity() {
        glMatrixValues.fill(0f)
        glMatrixValues[0] = 1f
        glMatrixValues[4] = 1f
        glMatrixValues[8] = 1f
        matrixBuffer.clear()
        matrixBuffer.put(glMatrixValues)
        matrixBuffer.position(0)
    }
}
