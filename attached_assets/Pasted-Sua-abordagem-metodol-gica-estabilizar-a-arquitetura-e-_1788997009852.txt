Snapshot Atômico do Giroscópio (Zero-Allocation)Para evitar que o renderer leia (x2, y1) de leituras assíncronas do sensor sem alocar um novo objeto Float2 ou Vector2 por frame, utilizamos uma referência imutável atualizada via classe de dados com substituição atômica (AtomicReference).Kotlinimport java.util.concurrent.atomic.AtomicReference

class GyroSensorManager : SensorEventListener {

    data class MotionState(val offsetX: Float, val offsetY: Float)

    private val currentState = AtomicReference(MotionState(0f, 0f))

    // Chamado na thread de sensores do Android
    override onSensorChanged(event: SensorEvent?) {
        event?.let {
            val rawX = it.values[0]
            val rawY = it.values[1]
            
            // Aplica filtro/dead-zone simples e atualiza atomicamente a referência inteira
            val filteredX = applyDeadZone(rawX)
            val filteredY = applyDeadZone(rawY)

            currentState.set(MotionState(filteredX, filteredY))
        }
    }

    // Chamado na GL Thread de forma thread-safe e sem locks bloqueantes
    fun getLatestState(): MotionState = currentState.get()

    private fun applyDeadZone(value: Float, threshold: Float = 0.005f): Float {
        return if (kotlin.math.abs(value) < threshold) 0f else value
    }

    override onAccuracyChanged(sensor: Sensor?, accuracy: Int) {}
}
2. EGL Lifecycle Manager e Threading RobustoO join(500) na main thread é um risco de ANR e leak de contexto OpenGL. A thread de renderização abaixo possui controle rigoroso do loop, cleanup na própria thread GL e destruição idempotente.Kotlinimport android.opengl.*
import android.view.SurfaceHolder

class WallpaperRenderThread(
    private val surfaceHolder: SurfaceHolder,
    private val textureProvider: TextureProvider
) : Thread("WallpaperRenderThread") {

    @Volatile
    private var isRunning = false

    @Volatile
    private var isPaused = false

    private var eglDisplay: EGLDisplay = EGL14.EGL_NO_DISPLAY
    private var eglContext: EGLContext = EGL14.EGL_NO_CONTEXT
    private var eglSurface: EGLSurface = EGL14.EGL_NO_SURFACE

    override fun run() {
        initEGL()
        initGLResources()

        isRunning = true
        var lastRenderTime = System.nanoTime()

        try {
            while (isRunning) {
                if (isPaused) {
                    synchronized(this) {
                        while (isPaused && isRunning) {
                            (this as java.lang.Object).wait()
                        }
                    }
                }

                if (!isRunning) break

                // Render loop
                renderFrame()

                // eglSwapBuffers lida com o VSync quando eglSwapInterval é 1
                if (!EGL14.eglSwapBuffers(eglDisplay, eglSurface)) {
                    val error = EGL14.eglGetError()
                    if (error == EGL14.EGL_CONTEXT_LOST) {
                        // Contexto perdido: recria recursos totalmente
                        recreateEGLContext()
                    }
                }
            }
        } finally {
            releaseGLResources()
            releaseEGL()
        }
    }

    private fun initEGL() {
        eglDisplay = EGL14.eglGetDisplay(EGL14.EGL_DEFAULT_DISPLAY)
        check(eglDisplay != EGL14.EGL_NO_DISPLAY) { "eglGetDisplay falhou" }

        val version = IntArray(2)
        check(EGL14.eglInitialize(eglDisplay, version, 0, version, 1)) { "eglInitialize falhou" }

        val configAttribs = intArrayOf(
            EGL14.EGL_RENDERABLE_TYPE, EGL14.EGL_OPENGL_ES2_BIT,
            EGL14.EGL_SURFACE_TYPE, EGL14.EGL_WINDOW_BIT, // EGL_WINDOW_BIT estritamente requerido
            EGL14.EGL_RED_SIZE, 8,
            EGL14.EGL_GREEN_SIZE, 8,
            EGL14.EGL_BLUE_SIZE, 8,
            EGL14.EGL_ALPHA_SIZE, 8,
            EGL14.EGL_NONE
        )

        val configs = arrayOfNulls<EGLConfig>(1)
        val numConfigs = IntArray(1)
        EGL14.eglChooseConfig(eglDisplay, configAttribs, 0, configs, 0, 1, numConfigs, 0)
        check(numConfigs[0] > 0) { "Nenhuma configuração EGL encontrada" }

        val contextAttribs = intArrayOf(
            EGL14.EGL_CONTEXT_CLIENT_VERSION, 2,
            EGL14.EGL_NONE
        )

        eglContext = EGL14.eglCreateContext(eglDisplay, configs[0], EGL14.EGL_NO_CONTEXT, contextAttribs, 0)
        check(eglContext != EGL14.EGL_NO_CONTEXT) { "eglCreateContext falhou" }

        eglSurface = EGL14.eglCreateWindowSurface(eglDisplay, configs[0], surfaceHolder, intArrayOf(EGL14.EGL_NONE), 0)
        check(eglSurface != EGL14.EGL_NO_SURFACE) { "eglCreateWindowSurface falhou" }

        check(EGL14.eglMakeCurrent(eglDisplay, eglSurface, eglSurface, eglContext)) { "eglMakeCurrent falhou" }

        // Sincronização VSync real no eglSwapBuffers
        EGL14.eglSwapInterval(eglDisplay, 1)
    }

    fun onPause() {
        isPaused = true
    }

    fun onResume() {
        synchronized(this) {
            isPaused = false
            (this as java.lang.Object).notifyAll()
        }
    }

    // Desligamento seguro e idempotente que NÃO bloqueia a main thread com join() prolongado
    fun requestStop() {
        isRunning = false
        onResume() // Acorda a thread se estiver pausada para que ela possa finalizar no seu próprio escopo
    }

    private fun initGLResources() { /* Compila Shaders, Carrega Texturas */ }
    private fun renderFrame() { /* Desenha Quad */ }
    private fun releaseGLResources() { /* glDeleteTextures, glDeleteProgram, glDeleteShader */ }

    private fun releaseEGL() {
        if (eglDisplay != EGL14.EGL_NO_DISPLAY) {
            EGL14.eglMakeCurrent(eglDisplay, EGL14.EGL_NO_SURFACE, EGL14.EGL_NO_SURFACE, EGL14.EGL_NO_CONTEXT)
            if (eglSurface != EGL14.EGL_NO_SURFACE) EGL14.eglDestroySurface(eglDisplay, eglSurface)
            if (eglContext != EGL14.EGL_NO_CONTEXT) EGL14.eglDestroyContext(eglDisplay, eglContext)
            EGL14.eglTerminate(eglDisplay)
        }
        eglDisplay = EGL14.EGL_NO_DISPLAY
        eglContext = EGL14.EGL_NO_CONTEXT
        eglSurface = EGL14.EGL_NO_SURFACE
    }

    private fun recreateEGLContext() {
        releaseGLResources()
        releaseEGL()
        initEGL()
        initGLResources()
    }
}

Shader Otimizado com Uniforms Deslocados e Cache de AtributosSubstituindo o recálculo de três matrizes de 16 floats por frame, passamos a matriz de projeção/escala estática uma única vez e variamos apenas vetores vec2 dinâmicos para o parallax.Vertex Shader (parallax.vert)OpenGL Shading Languageattribute vec2 a_Position;  // Local space quad coords [-1..1]
attribute vec2 a_TexCoord;  // Normal UV [0..1]

uniform mat4 u_StaticUVMatrix;

// Deslocamentos dinâmicos de parallax para evitar operações matriciais na CPU
uniform vec2 u_OffsetBack;
uniform vec2 u_OffsetMiddle;
uniform vec2 u_OffsetFore;

varying vec2 v_UVBack;
varying vec2 v_UVMiddle;
varying vec2 v_UVFore;

void main() {
    gl_Position = vec4(a_Position, 0.0, 1.0);
    
    // Transforma coordenada de tela estática para coordenada da textura
    vec2 baseUV = (u_StaticUVMatrix * vec4(a_TexCoord, 0.0, 1.0)).xy;

    v_UVBack   = baseUV + u_OffsetBack;
    v_UVMiddle = baseUV + u_OffsetMiddle;
    v_UVFore   = baseUV + u_OffsetFore;
}
Fragment Shader (parallax.frag)OpenGL Shading Languageprecision mediump float;

uniform sampler2D u_TexBack;
uniform sampler2D u_TexMiddle;
uniform sampler2D u_TexFore;

// Flags de ativação de camadas (0.0 se textureId == 0, 1.0 se ativa)
uniform float u_HasMiddleLayer;
uniform float u_HasForeLayer;

uniform float u_AlphaMiddle;
uniform float u_AlphaFore;

varying vec2 v_UVBack;
varying vec2 v_UVMiddle;
varying vec2 v_UVFore;

void main() {
    // Fundo SEMPRE opaco e assumido de base (Premultiplied Alpha)
    vec4 colBack = texture2D(u_TexBack, v_UVBack);
    colBack.a = 1.0; 

    vec4 finalColor = colBack;

    // Processa Camada Meio
    if (u_HasMiddleLayer > 0.5) {
        vec4 colMiddle = texture2D(u_TexMiddle, v_UVMiddle) * clamp(u_AlphaMiddle, 0.0, 1.0);
        // Blending Premultiplied Alpha manual no shader
        finalColor = colMiddle + finalColor * (1.0 - colMiddle.a);
    }

    // Processa Camada Frente
    if (u_HasForeLayer > 0.5) {
        vec4 colFore = texture2D(u_TexFore, v_UVFore) * clamp(u_AlphaFore, 0.0, 1.0);
        finalColor = colFore + finalColor * (1.0 - colFore.a);
    }

    gl_FragColor = finalColor;
}
4. Cache dos Locais dos Uniforms (Sem Consultas no Frame Loop)Kotlinclass ParallaxProgram(vertexSrc: String, fragmentSrc: String) {

    val programId: Int
    
    // Attribute locations
    val aPositionLoc: Int
    val aTexCoordLoc: Int

    // Uniform locations (consultados UMA ÚNICA VEZ na compilação do programa)
    val uStaticUVMatrixLoc: Int
    val uOffsetBackLoc: Int
    val uOffsetMiddleLoc: Int
    val uOffsetForeLoc: Int
    val uTexBackLoc: Int
    val uTexMiddleLoc: Int
    val uTexForeLoc: Int
    val uHasMiddleLayerLoc: Int
    val uHasForeLayerLoc: Int

    init {
        val vertShader = compileShader(GLES20.GL_VERTEX_SHADER, vertexSrc)
        val fragShader = compileShader(GLES20.GL_FRAGMENT_SHADER, fragmentSrc)
        
        programId = GLES20.glCreateProgram().also {
            GLES20.glAttachShader(it, vertShader)
            GLES20.glAttachShader(it, fragShader)
            GLES20.glLinkProgram(it)
            
            // Valida Link Status
            val linkStatus = IntArray(1)
            GLES20.glGetProgramiv(it, GLES20.GL_LINK_STATUS, linkStatus, 0)
            if (linkStatus[0] == 0) {
                val log = GLES20.glGetProgramInfoLog(it)
                GLES20.glDeleteProgram(it)
                error("Falha ao linkar GL Program: $log")
            }
        }

        // DELETA os objetos shader imediatamente após o link para liberar memória da GPU
        GLES20.glDeleteShader(vertShader)
        GLES20.glDeleteShader(fragShader)

        // Cache das Posições de Atributos e Uniforms
        aPositionLoc = GLES20.glGetAttribLocation(programId, "a_Position")
        aTexCoordLoc = GLES20.glGetAttribLocation(programId, "a_TexCoord")

        uStaticUVMatrixLoc = GLES20.glGetUniformLocation(programId, "u_StaticUVMatrix")
        uOffsetBackLoc     = GLES20.glGetUniformLocation(programId, "u_OffsetBack")
        uOffsetMiddleLoc   = GLES20.glGetUniformLocation(programId, "u_OffsetMiddle")
        uOffsetForeLoc     = GLES20.glGetUniformLocation(programId, "u_OffsetFore")

        uTexBackLoc        = GLES20.glGetUniformLocation(programId, "u_TexBack")
        uTexMiddleLoc      = GLES20.glGetUniformLocation(programId, "u_TexMiddle")
        uTexForeLoc        = GLES20.glGetUniformLocation(programId, "u_TexFore")

        uHasMiddleLayerLoc = GLES20.glGetUniformLocation(programId, "u_HasMiddleLayer")
        uHasForeLayerLoc   = GLES20.glGetUniformLocation(programId, "u_HasForeLayer")
    }

    private fun compileShader(type: Int, source: String): Int {
        val shader = GLES20.glCreateShader(type)
        GLES20.glShaderSource(shader, source)
        GLES20.glCompileShader(shader)

        val compiled = IntArray(1)
        GLES20.glGetShaderiv(shader, GLES20.GL_COMPILE_STATUS, compiled, 0)
        if (compiled[0] == 0) {
            val log = GLES20.glGetShaderInfoLog(shader)
            GLES20.glDeleteShader(shader)
            error("Erro ao compilar shader tipo $type: $log")
        }
        return shader
    }
}
