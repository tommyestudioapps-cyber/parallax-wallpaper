// Define a máxima precisão disponível no aparelho para evitar saltos/micro-stuttering no movimento
#ifdef GL_FRAGMENT_PRECISION_HIGH
precision highp float;
#else
precision mediump float;
#endif

// 1. Texturas enviadas do Android
uniform sampler2D u_TexBackground; // Foto inteira (opaca, cobre a tela)
uniform sampler2D u_TexMiddle;     // Recorte transparente 1
uniform sampler2D u_TexForeground; // Recorte transparente 2

// 2. Matrizes de Transformação 3x3 (Escala, Crop, Alinhamento e Parallax Shift)
uniform mat3 u_MatBack;
uniform mat3 u_MatMiddle;
uniform mat3 u_MatFore;

// 3. Opacidade global opcional de cada camada (ex: 0.0 a 1.0)
uniform float u_AlphaMiddle;
uniform float u_AlphaFore;

varying vec2 v_TexCoord;

// Função para amostragem dos recortes transparentes
vec4 sampleCutoutLayer(sampler2D tex, mat3 transformMat, float layerAlpha) {
    // Early Exit: Se a camada estiver desativada pelo usuário, pula a amostragem de textura
    if (layerAlpha <= 0.001) {
        return vec4(0.0);
    }

    // Transforma a coordenada UV da tela para a coordenada da textura
    vec3 transformedUV = transformMat * vec3(v_TexCoord, 1.0);
    vec2 uv = transformedUV.xy;

    // Se o movimento da camada sair dos limites do recorte, retorna transparente
    if (uv.x < 0.0 || uv.x > 1.0 || uv.y < 0.0 || uv.y > 1.0) {
        return vec4(0.0);
    }

    vec4 color = texture2D(tex, uv);
    
    // Premultiplied Alpha: multiplica tanto as cores RGB quanto o Alpha pela opacidade da camada
    return color * layerAlpha;
}

// Algoritmo de Blend Premultiplied Alpha (Source-Over rápido e exato)
vec4 blendPremultiplied(vec4 back, vec4 front) {
    return front + back * (1.0 - front.a);
}

void main() {
    // 1. Amostra a camada de FUNDO (Foto inteira opaca com Overscan)
    vec3 uvBack = u_MatBack * vec3(v_TexCoord, 1.0);
    // Usa clamp para garantir que o fundo nunca deixe vazios
    vec2 clampedUVBack = clamp(uvBack.xy, 0.0, 1.0);
    vec4 cBack = texture2D(u_TexBackground, clampedUVBack);
    cBack.a = 1.0; // Garante opacidade total no fundo

    // 2. Amostra as camadas da FRENTE e MEIO (Recortes Transparentes)
    vec4 cMiddle = sampleCutoutLayer(u_TexMiddle, u_MatMiddle, u_AlphaMiddle);
    vec4 cFore   = sampleCutoutLayer(u_TexForeground, u_MatFore, u_AlphaFore);

    // 3. Composição em ordem de profundidade: (Fundo + Meio) + Frente
    vec4 composite = blendPremultiplied(cBack, cMiddle);
    composite      = blendPremultiplied(composite, cFore);

    // Cor final do pixel no wallpaper (sempre opaca graças ao fundo)
    gl_FragColor = vec4(composite.rgb, 1.0);
}
