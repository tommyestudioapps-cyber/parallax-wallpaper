    <script>
        function chamarAtencao(botao) {
            botao.classList.remove('tremendo', 'laser');
            void botao.offsetWidth;

            botao.classList.add('tremendo');

            setTimeout(() => {
                botao.classList.remove('tremendo');
                botao.classList.add('laser');

                setTimeout(() => {
                    botao.classList.remove('laser');
                }, 950); // 0.3s (delay) + 0.6s (fade) = 0.9s → arredondei p/ 950ms

            }, 800);
        }
    </script>
