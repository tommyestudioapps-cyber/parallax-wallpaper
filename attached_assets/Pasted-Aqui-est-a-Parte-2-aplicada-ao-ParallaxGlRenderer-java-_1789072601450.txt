  public void updateSensorState(float sensorMotionX, float sensorMotionY) {
    if (surfaceWidth <= 0 || surfaceHeight <= 0) {
      motionX = 0f;
      motionY = 0f;
      return;
    }
    float compositionWidth = textureManager.getCompositionCanvasWidth();
    if (compositionWidth <= 0f) compositionWidth = DEFAULT_CANVAS_WIDTH;
    // Mesma normalização do Canvas: motionPixels = sensorX * multiplier * (W / compositionWidth).
    // Convertendo pixels -> NDC e deixando o `u_DepthFactor` multiplicar em runtime:
    //   offsetX = sensorX * 2 / compositionWidth
    //   offsetY = sensorY * 2 * surfaceWidth / (compositionWidth * surfaceHeight)
    // O Y precisa do aspect ratio porque o Canvas desloca em pixels do eixo X
    // e converte para NDC-Y usando a altura real da superfície.
    motionX = sensorMotionX * 2f / compositionWidth;
    motionY = sensorMotionY * 2f * surfaceWidth / (compositionWidth * surfaceHeight);
  }

  private void updateLayerMvp(int index) {
    ParallaxTextureManager.LayerData layer = textureManager.getLayerData(index);
    if (layer == null || layer.imageWidth <= 0f || layer.imageHeight <= 0f) {
      Matrix.setIdentityM(mMVPMatrix, 0);
      return;
    }

    float canvasWidth = surfaceWidth;
    float canvasHeight = surfaceHeight;
    float compositionWidth = textureManager.getCompositionCanvasWidth();
    if (compositionWidth <= 0f) compositionWidth = DEFAULT_CANVAS_WIDTH;
    float coordinateScale = canvasWidth / compositionWidth;

    // Espelha exatamente o cálculo do drawLayer() do Canvas.
    float fitScale = Math.max(canvasWidth / layer.imageWidth, canvasHeight / layer.imageHeight);
    float drawScale = fitScale * layer.scale;

    float left;
    float top;
    float width;
    float height;

    if (layer.hasSourceCrop) {
      width = layer.cropWidth * drawScale;
      height = layer.cropHeight * drawScale;
      left = (canvasWidth - layer.imageWidth * drawScale) / 2f
          + layer.cropOriginX * drawScale
          + layer.x * coordinateScale;
      top = (canvasHeight - layer.imageHeight * drawScale) / 2f
          + layer.cropOriginY * drawScale
          + layer.y * coordinateScale;
    } else {
      float sourceWidth = layer.sourceWidth > 0f ? layer.sourceWidth : layer.textureWidth;
      float sourceHeight = layer.sourceHeight > 0f ? layer.sourceHeight : layer.textureHeight;
      width = sourceWidth * drawScale;
      height = sourceHeight * drawScale;
      left = (canvasWidth - width) / 2f + layer.x * coordinateScale;
      top = (canvasHeight - height) / 2f + layer.y * coordinateScale;
    }

    // Centro do destino em pixels -> NDC.
    float centerPixelX = left + width / 2f;
    float centerPixelY = top + height / 2f;
    float centerNdcX = (centerPixelX / canvasWidth) * 2f - 1f;
    float centerNdcY = 1f - (centerPixelY / canvasHeight) * 2f;

    // Meia-extensão em NDC. O quad base vai de -1 a 1 (largura 2 em NDC).
    float scaleNdcX = width / canvasWidth;
    float scaleNdcY = height / canvasHeight;

    Matrix.setIdentityM(mModelMatrix, 0);
    Matrix.translateM(mModelMatrix, 0, centerNdcX, centerNdcY, 0f);
    Matrix.scaleM(mModelMatrix, 0, scaleNdcX, scaleNdcY, 1f);
    Matrix.multiplyMM(mMVPMatrix, 0, mProjMatrix, 0, mModelMatrix, 0);
  }

O que muda na prática
Parallax do OpenGL agora tem a mesma amplitude do Canvas (antes ficava ~10×–100× menor por causa da normalização errada).

Crop passa a ser renderizado geometricamente com a extensão correta (as UVs já estavam certas desde a Parte 1).

scale, x, y das camadas agora são aplicados no OpenGL exatamente como no Canvas.

canvasWidth da composição é respeitado em ambos os renderizadores.
