Substitua a classe interna inteira por:

java
  private static final class AlphaPrecisionGlState {
    private final int[] viewport;
    private final int framebufferBinding;
    private final int activeTexture;
    private final int textureBinding;
    private final int textureZeroBinding;
    private final int currentProgram;
    private final int arrayBufferBinding;
    private final boolean blendEnabled;
    private final int blendSourceRgb;
    private final int blendDestinationRgb;
    private final int blendSourceAlpha;
    private final int blendDestinationAlpha;
    private final int blendEquationRgb;
    private final int blendEquationAlpha;
    private final float[] clearColor;
    private final int textureUniform;
    private final float[] offset;
    private final float depthFactor;
    private final float[] uvTransform;
    private final float[] mvpMatrix;
    private final int[] scissorBox;
    private final boolean[] colorWriteMask;
    private final boolean scissorTestEnabled;
    private final boolean cullFaceEnabled;
    private final boolean depthTestEnabled;
    private final boolean stencilTestEnabled;
    private final VertexAttributeState position;
    private final VertexAttributeState texCoord;

    private AlphaPrecisionGlState(
        int[] viewport,
        int framebufferBinding,
        int activeTexture,
        int textureBinding,
        int textureZeroBinding,
        int currentProgram,
        int arrayBufferBinding,
        boolean blendEnabled,
        int blendSourceRgb,
        int blendDestinationRgb,
        int blendSourceAlpha,
        int blendDestinationAlpha,
        int blendEquationRgb,
        int blendEquationAlpha,
        float[] clearColor,
        int textureUniform,
        float[] offset,
        float depthFactor,
        float[] uvTransform,
        float[] mvpMatrix,
        int[] scissorBox,
        boolean[] colorWriteMask,
        boolean scissorTestEnabled,
        boolean cullFaceEnabled,
        boolean depthTestEnabled,
        boolean stencilTestEnabled,
        VertexAttributeState position,
        VertexAttributeState texCoord) {
      this.viewport = viewport;
      this.framebufferBinding = framebufferBinding;
      this.activeTexture = activeTexture;
      this.textureBinding = textureBinding;
      this.textureZeroBinding = textureZeroBinding;
      this.currentProgram = currentProgram;
      this.arrayBufferBinding = arrayBufferBinding;
      this.blendEnabled = blendEnabled;
      this.blendSourceRgb = blendSourceRgb;
      this.blendDestinationRgb = blendDestinationRgb;
      this.blendSourceAlpha = blendSourceAlpha;
      this.blendDestinationAlpha = blendDestinationAlpha;
      this.blendEquationRgb = blendEquationRgb;
      this.blendEquationAlpha = blendEquationAlpha;
      this.clearColor = clearColor;
      this.textureUniform = textureUniform;
      this.offset = offset;
      this.depthFactor = depthFactor;
      this.uvTransform = uvTransform;
      this.mvpMatrix = mvpMatrix;
      this.scissorBox = scissorBox;
      this.colorWriteMask = colorWriteMask;
      this.scissorTestEnabled = scissorTestEnabled;
      this.cullFaceEnabled = cullFaceEnabled;
      this.depthTestEnabled = depthTestEnabled;
      this.stencilTestEnabled = stencilTestEnabled;
      this.position = position;
      this.texCoord = texCoord;
    }

    private void restore(ParallaxGlRenderer renderer) {
      GLES20.glUseProgram(renderer.program);
      GLES20.glUniform1i(renderer.textureLocation, textureUniform);
      GLES20.glUniform2f(renderer.offsetLocation, offset[0], offset[1]);
      GLES20.glUniform1f(renderer.depthFactorLocation, depthFactor);
      GLES20.glUniform4f(
          renderer.uvTransformLocation,
          uvTransform[0],
          uvTransform[1],
          uvTransform[2],
          uvTransform[3]);
      GLES20.glUniformMatrix4fv(
          renderer.mvpMatrixLocation,
          1,
          false,
          mvpMatrix,
          0);

      GLES20.glBlendFuncSeparate(
          blendSourceRgb,
          blendDestinationRgb,
          blendSourceAlpha,
          blendDestinationAlpha);
      GLES20.glBlendEquationSeparate(blendEquationRgb, blendEquationAlpha);
      if (blendEnabled) {
        GLES20.glEnable(GLES20.GL_BLEND);
      } else {
        GLES20.glDisable(GLES20.GL_BLEND);
      }

      GLES20.glDisable(GLES20.GL_SCISSOR_TEST);
      GLES20.glDisable(GLES20.GL_CULL_FACE);
      GLES20.glDisable(GLES20.GL_DEPTH_TEST);
      GLES20.glDisable(GLES20.GL_STENCIL_TEST);
      GLES20.glColorMask(true, true, true, true);
      position.restore(renderer.positionLocation);
      texCoord.restore(renderer.texCoordLocation);
      GLES20.glBindBuffer(GLES20.GL_ARRAY_BUFFER, arrayBufferBinding);
      GLES20.glActiveTexture(activeTexture);
      GLES20.glBindTexture(GLES20.GL_TEXTURE_2D, textureBinding);
      GLES20.glActiveTexture(GLES20.GL_TEXTURE0);
      GLES20.glBindTexture(GLES20.GL_TEXTURE_2D, textureZeroBinding);
      GLES20.glActiveTexture(activeTexture);
      GLES20.glBindFramebuffer(GLES20.GL_FRAMEBUFFER, framebufferBinding);
      GLES20.glViewport(viewport[0], viewport[1], viewport[2], viewport[3]);
      GLES20.glClearColor(clearColor[0], clearColor[1], clearColor[2], clearColor[3]);
      GLES20.glScissor(scissorBox[0], scissorBox[1], scissorBox[2], scissorBox[3]);
      GLES20.glColorMask(
          colorWriteMask[0],
          colorWriteMask[1],
          colorWriteMask[2],
          colorWriteMask[3]);
      if (scissorTestEnabled) GLES20.glEnable(GLES20.GL_SCISSOR_TEST);
      if (cullFaceEnabled) GLES20.glEnable(GLES20.GL_CULL_FACE);
      if (depthTestEnabled) GLES20.glEnable(GLES20.GL_DEPTH_TEST);
      if (stencilTestEnabled) GLES20.glEnable(GLES20.GL_STENCIL_TEST);
      GLES20.glUseProgram(currentProgram);
    }
  }
